🔧 Step 2: server.js (FULL CODE)
const express = require('express');
const mongoose = require('mongoose');
const bodyParser = require('body-parser');

const app = express();

// Middleware
app.use(bodyParser.json());

// MongoDB Connection
mongoose.connect('mongodb://127.0.0.1:27017/bookstore')
.then(()=> console.log("MongoDB Connected"))
.catch(err => console.log(err));

// Schema
const bookSchema = new mongoose.Schema({
  title: String,
  author: String,
  price: Number,
  genre: String
});

const Book = mongoose.model('Book', bookSchema);


// ➤ 1. Add Book
app.post('/addBook', async (req,res)=>{
  try {
    const book = new Book(req.body);
    await book.save();
    res.json({ message: "Book Added" });
  } catch (err) {
    res.send(err);
  }
});


// ➤ 2. Get All Books
app.get('/books', async (req,res)=>{
  try {
    const data = await Book.find();
    res.json(data);
  } catch (err) {
    res.send(err);
  }
});


// ➤ 3. Update Book
app.put('/updateBook/:id', async (req,res)=>{
  try {
    await Book.findByIdAndUpdate(req.params.id, req.body);
    res.json({ message: "Book Updated" });
  } catch (err) {
    res.send(err);
  }
});


// ➤ 4. Delete Book
app.delete('/deleteBook/:id', async (req,res)=>{
  try {
    await Book.findByIdAndDelete(req.params.id);
    res.json({ message: "Book Deleted" });
  } catch (err) {
    res.send(err);
  }
});


// Start Server
app.listen(3000, ()=>{
  console.log("Server running at http://localhost:3000");
});
▶️ Run the Project

Option 1: Super Simple (HTML + Vanilla JS)

This is the fastest way to get a working UI.

1. Create a file: index.html
<!DOCTYPE html>
<html>
<head>
  <title>Bookstore</title>
</head>
<body>
  <h1>Bookstore</h1>

  <h2>Add Book</h2>
  <input id="title" placeholder="Title"><br>
  <input id="author" placeholder="Author"><br>
  <input id="price" placeholder="Price"><br>
  <input id="genre" placeholder="Genre"><br>
  <button onclick="addBook()">Add</button>

  <h2>Books</h2>
  <button onclick="getBooks()">Load Books</button>
  <ul id="bookList"></ul>

<script>
const API = "http://localhost:3000";

// Add book
async function addBook() {
  const book = {
    title: document.getElementById("title").value,
    authon: document.getElementById("author").value, // matches your schema typo
    price: document.getElementById("price").value,
    genre: document.getElementById("genre").value
  };

  await fetch(API + "/addBook", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(book)
  });

  alert("Book added!");
}

// Get books
async function getBooks() {
  const res = await fetch(API + "/books");
  const data = await res.json();

  const list = document.getElementById("bookList");
  list.innerHTML = "";

  data.forEach(book => {
    const li = document.createElement("li");
    li.innerHTML = `${book.title} - ${book.authon} - ₹${book.price}`;
    list.appendChild(li);
  });
}
</script>

</body>
</html>
2. Run it

Just open the file in your browser.

⚠️ If you get a CORS error, fix it in your backend:

npm install cors

Then add:

const cors = require('cors');
app.use(cors());
