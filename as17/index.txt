🔧 Step 3: server.js
const express = require('express');
const app = express();
const path = require('path');

const employees = require('./employees.json');

// API route
app.get('/api/employees', (req, res) => {
  res.json(employees);
});

// Serve static files
app.use(express.static(path.join(__dirname, 'public')));

// Start server
app.listen(3000, () => {
  console.log("Server running at http://localhost:3000");
});

<html>

  <style>

      border-collapse: collapse;
      width: 100%;
    }

      border: 1px solid #ccc;

      width: 80px;

      border-radius: 50%;
    }
  </style>

<body>

<h2>Employee Directory</h2>

<table id="empTable">
  <tr>
    <th>Image</th>
    <th>Name</th>
    <th>Designation</th>
    <th>Department</th>
    <th>Salary</th>
  </tr>
</table>

<script>
fetch('/api/employees')
  .then(res => res.json())
  .then(data => {

    let rows = '';

    data.forEach(emp => {
      rows += `
        <tr>
          <td><img src="${emp.image}"></td>
          <td>${emp.name}</td>
          <td>${emp.designation}</td>
          <td>${emp.department}</td>
          <td>₹${emp.salary}</td>
        </tr>
      `;
    });

    document.getElementById('empTable').innerHTML += rows;
  })
  .catch(err => console.log(err));
</script>

</body>
</html>
